cmake_minimum_required(VERSION 3.16)
project(ThreadDemo CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
file(GLOB_RECURSE DEMO_SOURCES ${CMAKE_SOURCE_DIR}/src/*.cpp)
add_executable(thread_demo ${DEMO_SOURCES})
target_include_directories(thread_demo PRIVATE ${CMAKE_SOURCE_DIR}/src)
target_link_libraries(thread_demo PRIVATE Threads::Threads)
